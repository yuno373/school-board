  <div class="admin">
    <h3>管理者設定</h3>
    <input type="password" id="adminPass" placeholder="現在のパスワード">
    <input type="password" id="newPass" placeholder="新しいパスワード">

<script>
const API = "https://photo-api.jaimely.workers.dev"; // ←唯希のURLに変更してね

async function upload() {
  const file = document.getElementById("fileInput").files[0];
  if (!file) return alert("ファイルを選んでね");

  const url = `${API}/upload?user=admin&filename=${encodeURIComponent(file.name)}`;

  const res = await fetch(url, {
    method: "POST",
    body: file
  });

  alert(await res.text());
  loadGallery();
}

async function loadGallery() {
  const res = await fetch(`${API}/list?user=admin`);
  const list = await res.json();

  const gallery = document.getElementById("gallery");
  gallery.innerHTML = "";

  list.forEach(url => {
    const img = document.createElement("img");
    img.src = url;
    gallery.appendChild(img);
  });
}

async function changePass() {
  const pass = document.getElementById("adminPass").value;
  const newPass = document.getElementById("newPass").value;

  const res = await fetch(`${API}/change-password?password=${pass}&new=${newPass}`);
  alert(await res.text());
}

loadGallery();
</script>
